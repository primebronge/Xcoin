name: Build APK

on:
  push:
    branches: [main, master]
  workflow_dispatch:        # Actions tab theke "Run workflow" button diye-o chalano jay

permissions:
  contents: write           # Release e APK upload korar jonno

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Set up JDK 17
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'

      - name: Set up Gradle 8.9
        uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: '8.9'

      - name: Accept Android SDK licenses
        run: yes | sdkmanager --licenses > /dev/null || true

      - name: Use own keystore (optional)
        env:
          KS_B64: ${{ secrets.KEYSTORE_BASE64 }}
          KS_PASS: ${{ secrets.KEYSTORE_PASSWORD }}
          KS_ALIAS: ${{ secrets.KEY_ALIAS }}
          KS_KEYPASS: ${{ secrets.KEY_PASSWORD }}
        run: |
          if [ -n "$KS_B64" ]; then
            echo "$KS_B64" | base64 -d > android/app/custom.keystore
            {
              echo "KEYSTORE_FILE=custom.keystore"
              echo "KEYSTORE_PASSWORD=$KS_PASS"
              echo "KEY_ALIAS=$KS_ALIAS"
              echo "KEY_PASSWORD=$KS_KEYPASS"
            } >> "$GITHUB_ENV"
            echo "Custom keystore in use"
          else
            echo "Default sideload keystore in use"
          fi

      - name: Build release APK
        working-directory: android
        run: gradle assembleRelease -PversionCode=${{ github.run_number }} --stacktrace

      - name: Collect APK
        run: |
          mkdir -p dist
          cp android/app/build/outputs/apk/release/app-release.apk dist/Ludo-King.apk
          ls -lh dist

      - name: Upload APK (Actions artifact)
        uses: actions/upload-artifact@v4
        with:
          name: Ludo-King-APK
          path: dist/Ludo-King.apk
          if-no-files-found: error

      - name: Publish release (direct download link)
        uses: softprops/action-gh-release@v2
        with:
          tag_name: build-${{ github.run_number }}
          name: Ludo King build ${{ github.run_number }}
          target_commitish: ${{ github.sha }}
          files: dist/Ludo-King.apk
